# StarHermit engine script game: one package, played natively by the
# StarHermit client and in browsers by the engine's web runtime. Browsers open
# launch=index.html, the loader `npm run package` writes (it pins the engine
# on wiki.starhermit.com by hash; never edit it). Re-run `npm run package`
# after every change and commit index.html + starhermit-files.json.
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.23.0
engine.minor=20
engine.capabilities=platform,storage,clipboard

# Keyboard controls (players rebind keys and pads in the runtime's Options).
# The arena is also played with the pointer or touch: drag on the arena to
# steer the paddle, tap it to serve; the HUD has Serve and Pause buttons.
# Shared-screen 2P: the second player uses A / D and W. (The browser build's
# p2Left / p2Right / p2Serve actions are p2left / p2right / p2serve here:
# action names are lowercase.)
control.left=ArrowLeft | Move left
control.right=ArrowRight | Move right
control.serve=Space | Serve
control.pause=Escape | Pause
control.undo=KeyU | Undo (practice)
control.camera=KeyC | Toggle camera
control.hint=KeyH | Hint
control.p2left=KeyA | Player 2 left (shared screen)
control.p2right=KeyD | Player 2 right (shared screen)
control.p2serve=KeyW | Player 2 serve (shared screen)
engine.group.left=Paddle
engine.group.pause=Match
engine.group.p2left=Shared screen

# Gamepad defaults (the browser build's mapping: the left stick or D-pad
# steers, A serves, Start pauses; X hints, Y undoes, RB toggles the camera).
engine.gamepad.left=LeftStickLeft+DpadLeft
engine.gamepad.right=LeftStickRight+DpadRight
engine.gamepad.serve=South
engine.gamepad.pause=Start
engine.gamepad.hint=West
engine.gamepad.undo=North
engine.gamepad.camera=RightShoulder
